// LeetCode 3249 - Count the Number of Good Nodes
// https://leetcode.com/problems/count-the-number-of-good-nodes/

var countGoodNodes = function(edges) {
    const n = edges.length + 1;
    const g = Array.from({length: n}, () => []);
    for (const e of edges) { g[e[0]].push(e[1]); g[e[1]].push(e[0]); }
    let ans = 0;
    const dfs = (a, fa) => {
        let pre = -1, cnt = 1, ok = 1;
        for (const b of g[a]) {
            if (b !== fa) {
                const cur = dfs(b, a);
                cnt += cur;
                if (pre < 0) pre = cur;
                else if (pre !== cur) ok = 0;
            }
        }
        ans += ok;
        return cnt;
    };
    dfs(0, -1);
    return ans;
};
