// LeetCode 3260 - Find the Largest Palindrome Divisible by K
// https://leetcode.com/problems/find-the-largest-palindrome-divisible-by-k/

var largestPalindrome = function(n, k) {
    const repeat = (c, n) => Array(n).fill(c);
    let digits = repeat('9', n);
    const half = Math.floor((n + 1) / 2);
    const mod7 = (s) => {
        let r = 0;
        for (let i = 0; i < s.length; i++) r = (r * 10 + (s.charCodeAt(i) - 48)) % 7;
        return r;
    };
    const largestPal7 = (n) => {
        const halfLen = Math.floor((n + 1) / 2);
        const half = repeat('9', halfLen);
        for (;;) {
            const pal = new Array(n);
            for (let i = 0; i < halfLen; i++) pal[i] = half[i];
            for (let i = 0; i < Math.floor(n / 2); i++) pal[n - 1 - i] = pal[i];
            if (mod7(pal.join('')) === 0) return pal.join('');
            let idx = halfLen - 1;
            while (idx >= 0 && half[idx] === '0') { half[idx] = '9'; idx--; }
            if (idx < 0) break;
            half[idx] = String.fromCharCode(half[idx].charCodeAt(0) - 1);
        }
        return '';
    };
    switch (k) {
        case 1: case 3: case 9:
            return digits.join('');
        case 2:
            digits[0] = digits[n - 1] = '8';
            return digits.join('');
        case 4:
            if (n === 1) return '8';
            digits[0] = digits[1] = digits[n - 1] = digits[n - 2] = '8';
            return digits.join('');
        case 5:
            digits[0] = digits[n - 1] = '5';
            return digits.join('');
        case 8:
            if (n <= 2) return repeat('8', n).join('');
            digits[0] = digits[1] = digits[2] = '8';
            digits[n - 1] = digits[n - 2] = digits[n - 3] = '8';
            return digits.join('');
        case 6: {
            if (n === 1) return '6';
            digits[0] = digits[n - 1] = '8';
            const sum = 16 + 9 * (n - 2);
            const need = sum % 3;
            if (need !== 0) {
                const pos = half - 1;
                digits[pos] = String.fromCharCode(digits[pos].charCodeAt(0) - need);
                if (n % 2 === 0 || pos !== n - 1 - pos) digits[n - 1 - pos] = digits[pos];
            }
            return digits.join('');
        }
        case 7:
            return largestPal7(n);
        default:
            return digits.join('');
    }
};
