// LeetCode 3203 - Find Minimum Diameter After Merging Two Trees
// https://leetcode.com/problems/find-minimum-diameter-after-merging-two-trees/

var minimumDiameterAfterMerge = function(edges1, edges2) {
    let ans = 0, a = 0, g;
    const dfs = (i, fa, t) => {
        for (const j of g[i]) if (j !== fa) dfs(j, i, t + 1);
        if (ans < t) { ans = t; a = i; }
    };
    const treeDiameter = (edges) => {
        const n = edges.length + 1;
        g = Array.from({length: n}, () => []);
        for (const e of edges) { g[e[0]].push(e[1]); g[e[1]].push(e[0]); }
        ans = 0; a = 0;
        dfs(0, -1, 0);
        dfs(a, -1, 0);
        return ans;
    };
    const d1 = treeDiameter(edges1);
    const d2 = treeDiameter(edges2);
    return Math.max(d1, d2, Math.floor((d1 + 1) / 2) + Math.floor((d2 + 1) / 2) + 1);
};
