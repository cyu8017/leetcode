// LeetCode 3676 - Count Bowl Subarrays
// https://leetcode.com/problems/count-bowl-subarrays/

import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;

class Solution {
    public long bowlSubarrays(int[] nums) {
        int n = nums.length;
        long ans = 0;
        int[] ngr = new int[n], ngl = new int[n];
        Arrays.fill(ngr, -1);
        Arrays.fill(ngl, -1);
        List<Integer> stack = new ArrayList<>();
        for (int i = n - 1; i >= 0; i--) {
            while (!stack.isEmpty() && nums[stack.get(stack.size() - 1)] < nums[i])
                stack.remove(stack.size() - 1);
            if (!stack.isEmpty()) ngr[i] = stack.get(stack.size() - 1);
            stack.add(i);
        }
        stack.clear();
        for (int i = 0; i < n; i++) {
            while (!stack.isEmpty() && nums[stack.get(stack.size() - 1)] < nums[i])
                stack.remove(stack.size() - 1);
            if (!stack.isEmpty()) ngl[i] = stack.get(stack.size() - 1);
            stack.add(i);
        }
        for (int i = 0; i < n; i++) {
            if (ngr[i] != -1 && ngr[i] - i >= 2) ans++;
            if (ngl[i] != -1 && i - ngl[i] >= 2) ans++;
        }
        return ans;
    }
}
