// LeetCode 3717 - Minimum Operations to Make the Array Beautiful
// https://leetcode.com/problems/minimum-operations-to-make-the-array-beautiful/

import java.util.HashMap;
import java.util.Map;

class Solution {
    public int minOperations(int[] nums) {
        Map<Integer, Integer> f = new HashMap<>();
        f.put(nums[0], 0);
        for (int i = 1; i < nums.length; i++) {
            int x = nums[i];
            Map<Integer, Integer> g = new HashMap<>();
            for (Map.Entry<Integer, Integer> e : f.entrySet()) {
                int pre = e.getKey(), s = e.getValue();
                int cur = (x + pre - 1) / pre * pre;
                while (cur <= 100) {
                    int val = s + (cur - x);
                    Integer old = g.get(cur);
                    if (old == null || old > val) g.put(cur, val);
                    cur += pre;
                }
            }
            f = g;
        }
        int ans = Integer.MAX_VALUE;
        for (int v : f.values()) ans = Math.min(ans, v);
        return ans;
    }
}
