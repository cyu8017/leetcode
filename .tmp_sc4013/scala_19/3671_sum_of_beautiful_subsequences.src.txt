// LeetCode 3671 - Sum of Beautiful Subsequences
// https://leetcode.com/problems/sum-of-beautiful-subsequences/

import java.util.ArrayList;
import java.util.Collections;
import java.util.List;

class Solution {
    public int totalBeauty(int[] nums) {
        final int MOD = 1_000_000_007;
        int mx = 0;
        for (int v : nums) if (v > mx) mx = v;
        @SuppressWarnings("unchecked")
        List<Integer>[] pos = new ArrayList[mx + 1];
        for (int i = 0; i <= mx; i++) pos[i] = new ArrayList<>();
        for (int i = 0; i < nums.length; i++) pos[nums[i]].add(i);
        int[] cnt = new int[mx + 1];
        for (int g = 1; g <= mx; g++) {
            List<Integer> seq = new ArrayList<>();
            for (int m = g; m <= mx; m += g) seq.addAll(pos[m]);
            if (seq.isEmpty()) continue;
            Collections.sort(seq);
            int ways = 1;
            for (int i = 0; i < seq.size(); i++) ways = (int) ((ways * 2L) % MOD);
            cnt[g] = (ways - 1 + MOD) % MOD;
        }
        int ans = 0;
        for (int g = mx; g >= 1; g--) {
            for (int m = 2 * g; m <= mx; m += g)
                cnt[g] = (cnt[g] - cnt[m] + MOD) % MOD;
            ans = (int) ((ans + 1L * cnt[g] * g) % MOD);
        }
        return ans;
    }
}
